CREATE DATABASE library;

USE library;

CREATE TABLE users (
  id int PRIMARY KEY AUTO_INCREMENT,
  name varchar(60) NOT NULL,
  fullname varchar(120) NOT NULL,
  email varchar(120) NOT NULL,
  phone varchar(12) DEFAULT '-',
  password varchar(265) NOT NULL,
  role enum('user','staff','admin') DEFAULT NULL,
  status enum('unverified','verified','deleted') DEFAULT 'unverified'
);

CREATE TABLE  personal_account (
  id int PRIMARY KEY NOT NULL AUTO_INCREMENT,
  id_user int NOT NULL,
  address varchar(265) NOT NULL,
  identify_type varchar(20) NOT NULL,
  identify_image varchar(265) NOT NULL,
  profile_image varchar(265) DEFAULT 'https://res.cloudinary.com/ddiulakke/image/upload/v1747055039/vecteezy_profile-icon-design-vector_5544718_cje74w.jpg',
  status_user enum('student','worker') NOT NULL,
  created_at bigint NOT NULL,
  updated_at bigint NOT NULL,
  status enum('success','fail') NOT NULL DEFAULT 'fail',
  FOREIGN KEY (id_user) REFERENCES users (id)
);

CREATE TABLE  books(
    id INT PRIMARY KEY AUTO_INCREMENT,
    isbn_10 varchar(10) DEFAULT NULL,
    isbn_13 varchar(13) DEFAULT NULL,
    title varchar(265) NOT NULL,
    description text NOT NULL,
    author varchar(125) NOT NULL,
    publisher varchar(265) NOT NULL,
    publication_year BIGINT NOT NULL,
    edition_number INT default 1,
    total_pages int NOT NULL,
    category varchar(265) NOT NULL,
    language varchar(255) DEFAULT 'indonesian',
    receiver_date BIGINT NOT NULL,
    clsn_id int NOT NULL,
    image_url varchar(255) NOT NULL,
    status enum('public','private'),
    FOREIGN KEY(clsn_id) REFERENCES classification_numbers(id)
);

CREATE TABLE  book_copys (
  id varchar(40) PRIMARY KEY NOT NULL,
  id_book int NOT NULL,
  call_number varchar(60) DEFAULT NULL,
  status enum('available','borrowed','lost','damaged','under repair') DEFAULT NULL,
  loan_date bigint DEFAULT NULL,
  due_date bigint DEFAULT NULL,
  physical_condition varchar(40) DEFAULT 'good',
  FOREIGN KEY(id_book) REFERENCES books(id)
);

CREATE TABLE  classification_numbers (
  id int PRIMARY KEY,
  classification varchar(265) NOT NULL,
  status enum('active','inactive') NOT NULL DEFAULT 'active'
);

     CREATE TABLE return_book(
        id int primary key auto_increment,
        id_user int not null,
        id_book int not null,
        return_condition enum('good','normal','damaged') not null,
        return_at bigint not null,
        foreign key (id_user) references users(id),
        foreign key (id_book) references books(id)
    );

    CREATE TABLE borrowed_books(
        id INT PRIMARY KEY AUTO_INCREMENT,
        id_user int not null,
        id_book_copy varchar(40) not null,
        book_condition enum('good','damaged') NOT NULL,
        borrowed_at BIGINT NOT NULL,
        return_condition enum('good','damaged','lost') NOT NULL,
        status enum('wait','borrowed','returned','fail') NOT NULL DEFAULT 'wait',
        return_at BIGINT,
        FOREIGN KEY(id_user) references users(id),
        FOREIGN KEY(id_book_copy) references book_copys(id)
    );

    CREATE TABLE action_logs (
        id int PRIMARY KEY AUTO_INCREMENT,
        user_id int NOT NULL,
        action varchar(265) NOT NULL,
        action_at bigint DEFAULT NULL,
        FOREIGN KEY(user_id) REFERENCES users(id)
    );
